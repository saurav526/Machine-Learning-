from sklearn.datasets import load_iris
👉 Imports the Iris dataset which is a built-in dataset in scikit-learn.

from sklearn.model_selection import train_test_split
👉 Used to split data into training and testing sets.

from sklearn.svm import SVC
👉 Imports Support Vector Classifier (SVM) algorithm.

from sklearn.metrics import accuracy_score
👉 Imports function to calculate accuracy of the model.

data = load_iris()
👉 Loads the Iris dataset into the variable data.

X = data.data
👉 Stores input features (sepal length, width, etc.) in X.

y = data.target
👉 Stores output labels (classes) in y.

X_train, X_test, y_train, y_test = train_test_split(
    X, y, test_size=0.3, random_state=1
)
👉 Splits data into:

70% training data

30% testing data
random_state=1 ensures same output every time.

svm_model = SVC(kernel='linear')
👉 Creates an SVM model using a linear kernel.

svm_model.fit(X_train, y_train)
👉 Trains the SVM model using training data.

y_pred = svm_model.predict(X_test)
👉 Predicts output for test data.

print("SVM Accuracy:", accuracy_score(y_test, y_pred))
👉 Calculates and prints accuracy of the SVM model.
